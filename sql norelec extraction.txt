DROP TABLE IF EXISTS ventes;

CREATE TABLE ventes (
vente_id INTEGER PRIMARY KEY,
date_vente TEXT NOT NULL,
produit_id INTEGER,
client_id INTEGER,
quantite INTEGER,
prix_unitaire REAL,
FOREIGN KEY (produit_id) REFERENCES produits(produit_id),
FOREIGN KEY (client_id) REFERENCES clients(client_id)
);

SELECT
    v.date_vente,
    c.nom_client,
    p.nom_produit,
    v.quantite,
    v.prix_unitaire,
    ROUND(v.quantite*v.prix_unitaire,2) AS montant_total
FROM ventes v
JOIN clients c ON v.client_id = c.client_id
JOIN produits p ON  v.produit_id = p.produit_id;

SELECT 
    c.nom_client,
    SUM(v.quantite * v.prix_unitaire) AS ca_total
FROM ventes v
JOIN clients c ON v.client_id = c.client_id
GROUP BY c.nom_client
ORDER BY ca_total DESC;

SELECT
strftime('%Y-%m', v.date_vente) AS mois,
ROUND(SUM(v.quantite* v.prix_unitaire),2) AS ca_total
FROM ventes v
GROUP BY mois
ORDER BY mois;

DROP TABLE IF EXISTS budget;

	  ROUND(b.ca_budget, 2) AS ca_budget,
	  ROUND(r.ca_reel - b.ca_budget, 2) AS ecart
	  FROM
	  (SELECT strftime('%Y-%m', v.date_vente) AS mois, SUM(quantite * prix_unitaire) AS ca_reel
